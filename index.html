using UnityEngine; using UnityEngine.SceneManagement; using UnityEngine.UI; public class AmmarYasserGame : MonoBehaviour { // ========================================================= // هذا السكربت الواحد يدير اللعبة كلها // ========================================================= public static AmmarYasserGame Instance; [Header("PLAYER")] public CharacterController player; public float forwardSpeed = 8f; public float laneDistance = 3f; public float laneChangeSpeed = 12f; public float jumpForce = 8f; public float gravity = -25f; [Header("SPAWNING")] public GameObject trainPrefab; public GameObject coinPrefab; public float spawnDistance = 80f; public float trainSpawnTime = 2f; public float coinSpawnTime = 1f; [Header("UI")] public Text scoreText; public Text coinText; public Text gameOverText; public Button restartButton; [Header("SETTINGS")] public int startingLane = 1; public float minimumSwipe = 50f; private int currentLane; private float verticalVelocity; private int score; private int coins; private float trainTimer; private float coinTimer; private float scoreTimer; private bool gameOver; private Vector2 touchStart; private bool touching; void Awake() { Instance = this; } void Start() { Time.timeScale = 1f; currentLane = startingLane; score = 0; coins = 0; gameOver = false; if (gameOverText != null) gameOverText.gameObject.SetActive(false); if (restartButton != null) restartButton.gameObject.SetActive(false); UpdateUI(); } void Update() { if (gameOver) return; PlayerMovement(); MobileControls(); SpawnObjects(); UpdateScore(); } // ========================================================= // حركة اللاعب // ========================================================= void PlayerMovement() { float targetX = (currentLane - 1) * laneDistance; Vector3 position = player.transform.position; float newX = Mathf.Lerp( position.x, targetX, laneChangeSpeed * Time.deltaTime ); if (player.isGrounded && verticalVelocity < 0) { verticalVelocity = -2f; } verticalVelocity += gravity * Time.deltaTime; Vector3 movement = new Vector3( newX - position.x, verticalVelocity, forwardSpeed ); player.Move(movement * Time.deltaTime); } // ========================================================= // تغيير المسار // ========================================================= void ChangeLane(int direction) { currentLane += direction; currentLane = Mathf.Clamp( currentLane, 0, 2 ); } // ========================================================= // القفز // ========================================================= void Jump() { if (player.isGrounded) { verticalVelocity = jumpForce; } } // ========================================================= // الانزلاق // ========================================================= void Slide() { player.transform.localScale = new Vector3(1f, 0.5f, 1f); CancelInvoke(nameof(StopSlide)); Invoke( nameof(StopSlide), 0.5f ); } void StopSlide() { player.transform.localScale = Vector3.one; } // ========================================================= // التحكم بالهاتف // ========================================================= void MobileControls() { if (Input.touchCount == 0) return; Touch touch = Input.GetTouch(0); if (touch.phase == TouchPhase.Began) { touchStart = touch.position; touching = true; } if ( touch.phase == TouchPhase.Ended && touching ) { Vector2 touchEnd = touch.position; Vector2 swipe = touchEnd - touchStart; touching = false; if (swipe.magnitude < minimumSwipe) return; if ( Mathf.Abs(swipe.x) > Mathf.Abs(swipe.y) ) { if (swipe.x > 0) ChangeLane(1); else ChangeLane(-1); } else { if (swipe.y > 0) Jump(); else Slide(); } } } // ========================================================= // لوحة المفاتيح للتجربة على الكمبيوتر // ========================================================= void KeyboardControls() { if ( Input.GetKeyDown(KeyCode.LeftArrow) || Input.GetKeyDown(KeyCode.A) ) { ChangeLane(-1); } if ( Input.GetKeyDown(KeyCode.RightArrow) || Input.GetKeyDown(KeyCode.D) ) { ChangeLane(1); } if ( Input.GetKeyDown(KeyCode.UpArrow) || Input.GetKeyDown(KeyCode.W) || Input.GetKeyDown(KeyCode.Space) ) { Jump(); } if ( Input.GetKeyDown(KeyCode.DownArrow) || Input.GetKeyDown(KeyCode.S) ) { Slide(); } } // ========================================================= // إنشاء القطارات والعملات // ========================================================= void SpawnObjects() { trainTimer += Time.deltaTime; coinTimer += Time.deltaTime; if (trainTimer >= trainSpawnTime) { SpawnTrain(); trainTimer = 0f; trainSpawnTime = Mathf.Max( 0.7f, trainSpawnTime - 0.01f ); } if (coinTimer >= coinSpawnTime) { SpawnCoin(); coinTimer = 0f; } } void SpawnTrain() { if (trainPrefab == null) return; int lane = Random.Range(0, 3); float x = (lane - 1) * laneDistance; Vector3 spawnPosition = new Vector3( x, 0f, player.transform.position.z + spawnDistance ); GameObject train = Instantiate( trainPrefab, spawnPosition, Quaternion.identity ); train.tag = "Train"; } void SpawnCoin() { if (coinPrefab == null) return; int lane = Random.Range(0, 3); float x = (lane - 1) * laneDistance; Vector3 spawnPosition = new Vector3( x, 1.2f, player.transform.position.z + spawnDistance ); GameObject coin = Instantiate( coinPrefab, spawnPosition, Quaternion.identity ); coin.AddComponent(); } // ========================================================= // النقاط // ========================================================= void UpdateScore() { scoreTimer += Time.deltaTime; if (scoreTimer >= 0.1f) { score++; scoreTimer = 0f; UpdateUI(); } } void AddCoin() { coins++; score += 10; UpdateUI(); } void UpdateUI() { if (scoreText != null) { scoreText.text = "النقاط: " + score; } if (coinText != null) { coinText.text = "العملات: " + coins; } } // ========================================================= // الاصطدام // ========================================================= void OnControllerColliderHit( ControllerColliderHit hit ) { if ( hit.collider.CompareTag("Train") || hit.collider.CompareTag("Obstacle") ) { GameOver(); } } // ========================================================= // نهاية اللعبة // ========================================================= void GameOver() { if (gameOver) return; gameOver = true; if (gameOverText != null) { gameOverText.gameObject.SetActive(true); gameOverText.text = "عمار ياسر\n" + "انتهت اللعبة\n" + "النقاط: " + score; } if (restartButton != null) { restartButton.gameObject.SetActive(true); } Time.timeScale = 0f; } // ========================================================= // إعادة اللعبة // ========================================================= public void RestartGame() { Time.timeScale = 1f; SceneManager.LoadScene( SceneManager.GetActiveScene().buildIndex ); } // ========================================================= // سكربت العملة داخل نفس الملف // ========================================================= public class CoinObject : MonoBehaviour { private void Update() { transform.Rotate( 0f, 180f * Time.deltaTime, 0f ); if ( AmmarYasserGame.Instance != null && !AmmarYasserGame.Instance.gameOver ) { float playerZ = AmmarYasserGame.Instance .player.transform.position.z; if ( transform.position.z < playerZ - 20f ) { Destroy(gameObject); } } } private void OnTriggerEnter( Collider other ) { if ( other.CompareTag("Player") ) { AmmarYasserGame.Instance .AddCoin(); Destroy(gameObject); } } } }
